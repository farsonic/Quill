# mac/traps.txt - the Toolbox calls Quill's Mac build makes beyond Riptide's (riptide
# src/mac/app/traps.txt, whose format this is: trap result Name(type name:kind, ...)). Makefile.mac
# joins the two and has riptide's tools/mac_traps.py make the glue. GPL-2.0-only, as Quill.
# (CopyBits, FrameOval and PaintOval, which the page view and the strip use, are Riptide's own in its
# octavo-ribbon branch, for its graphics.library calls: listed there, not again here.)
#
# Controls: the scroll bars drawn after an update
A96D - Draw1Control(void *ctl:l)
# Events: whether keys are waiting (typing is laid out once they stop)
A971 b EventAvail(int mask:w, void *ev:l)
# Resources: the System's region, for the interface language (mac/qm_lang.c)
A994 w CurResFile()
A998 - UseResFile(int ref:w)
A81F p Get1Resource(long type:l, int id:w)
# Dialogs: the text fields of Quill's own dialogs (mac/qm_dialog.c)
A990 - GetIText(void *h:l, unsigned char *s:l)
A98F - SetIText(void *h:l, const unsigned char *s:l)
A97E - SelIText(void *dlg:l, int item:w, int start:w, int end:w)
# QuickDraw: the page's text drawn into a one-bit bitmap of Octavo's own (mac/qm_qdtext.c)
A86F - OpenPort(void *port:l)
A875 - SetPortBits(const void *bm:l)
A876 - PortSize(int width:w, int height:w)
